#include <bits/stdc++.h>
using namespace std;

int triplet(vector<int> &a){
    int ans =0;
    for(int j=1; j<a.size()-1; j++){
        int lc=0, rc=0;
        for(int i=0; i<j; i++){
            if(a[i]<a[j]) lc++;
        }
        for(int k=j+1; k<a.size(); k++){
            if(a[k]>a[j]) rc++;
        }
        ans += lc*rc;
    }
    return ans;
}

int main(){

    vector<int> a = {2, 3 ,8 ,4 ,6, 10, 5, 12};
    cout << triplet(a);
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> transpose(vector<vector<int>> &mat){
    int m = mat.size();
    int n = mat[0].size(); // m x n
    vector<int> v;
    for(int i=0; i<m; i++){
        v.push_back(0);
    }
    vector<vector<int>> T;
    for(int i=0; i<n; i++){
        T.push_back(v);
    }

    for(int i=0; i<n; i++){
        for(int j=0; j<m; j++){
            T[i][j] = mat[j][i];
        }
    }
    return T;
}

int main(){
    vector<int> a = {10,20,30,40};
    vector<int> b = {50,60,80,90};
    vector<int> c = {2, 3, 4,56};

    vector<vector<int>> v;
    v.push_back(a);
    v.push_back(b);
    v.push_back(c);

    vector<vector<int>> ans = transpose(v);
    for(int i=0; i<ans.size(); i++){
        for(int j=0; j<ans[i].size(); j++){
            cout << ans[i][j] << " "; 
        }
        cout << "\n";
    }
      
    
    return 0;
}

#include <bits/stdc++.h>
using namespace std;

vector<vector<int>> rotate(vector<vector<int>> &mat){
    int n = mat.size(); // n x n
    for(int i=0; i<n; i++){
        for(int j=i+1; j<n; j++){
            swap(mat[i][j], mat[j][i]);
        }
    }
    for(int i=0; i<n; i++){
        reverse(mat[i].begin(), mat[i].end());
    }
    return mat;
}

int main(){
    vector<int> a = {10,20,30};
    vector<int> b = {50,60,80};
    vector<int> c = {2, 3, 4};

    vector<vector<int>> v;
    v.push_back(a);
    v.push_back(b);
    v.push_back(c);

    vector<vector<int>> ans = rotate(v);
    for(int i=0; i<ans.size(); i++){
        for(int j=0;j<ans.size(); j++){
            cout<< ans[i][j] << " ";
        }
        cout << "\n";
    }
    
    return 0;
}
